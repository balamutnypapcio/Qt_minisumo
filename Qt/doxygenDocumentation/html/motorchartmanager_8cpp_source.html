221void MotorChartManager::enforceAxisLimits(qreal min, qreal max)
222{
223 bool needAdjustment = false;
224 qreal newMin = min;
225 qreal newMax = max;
226 qreal rangeSize = max - min;

228 if (min < 0) {

237 needAdjustment = true;
